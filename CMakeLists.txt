cmake_minimum_required(VERSION 3.10)
project(nds-rom-reader C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(ndsrom src/ndsrom.c src/ndscomp.c src/ndsgfx.c src/ndspng.c src/ndscell.c src/ndsanim.c src/ndsgif.c src/ndstext.c src/ndssdat.c src/ndstex.c src/ndsmodel.c src/ndsrender.c src/ndsgltf.c src/ndsjnt.c)
target_include_directories(ndsrom PUBLIC include)
if(UNIX)
  target_link_libraries(ndsrom PUBLIC m)
endif()
if(NOT MSVC)
  target_compile_options(ndsrom PRIVATE -Wall -Wextra)
endif()

add_executable(ndsviewer viewer/main.c)
target_link_libraries(ndsviewer PRIVATE ndsrom)
if(NOT MSVC)
  target_compile_options(ndsviewer PRIVATE -Wall -Wextra)
endif()

enable_testing()
add_executable(test_gfx tests/test_gfx.c)
target_link_libraries(test_gfx PRIVATE ndsrom)
add_test(NAME gfx COMMAND test_gfx)
add_executable(test_audio_text tests/test_audio_text.c)
target_link_libraries(test_audio_text PRIVATE ndsrom)
add_test(NAME audio_text COMMAND test_audio_text)
add_executable(test_model tests/test_model.c)
target_link_libraries(test_model PRIVATE ndsrom m)
add_test(NAME model COMMAND test_model)
